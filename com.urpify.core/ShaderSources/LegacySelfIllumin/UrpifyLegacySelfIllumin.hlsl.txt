// Urpify: Built-in's Legacy Self-Illumin shaders under URP (fix after Unity's converter, lab LEG03).
// Unity's converter maps these materials to Simple Lit, whose emission is a texture of its own, and the glow is lost.
// Here the emission is the albedo times the alpha of _Illum, and the lighting is Built-in's Lambert (diffuse) or
// Blinn-Phong (specular, exponent _Shininess * 128, scaled by the gloss in the albedo's alpha), with the ambient from
// spherical harmonics, every URP light and its shadows. Written on URP's own ShaderLibrary only (URP 14 and 17).
// Each shader defines URPIFY_BUMP (normal map) and URPIFY_SPECULAR (Blinn-Phong) before including this file.
#ifndef URPIFY_LEGACY_SELF_ILLUMIN_INCLUDED
#define URPIFY_LEGACY_SELF_ILLUMIN_INCLUDED

#include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
#include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"

TEXTURE2D(_MainTex);  SAMPLER(sampler_MainTex);
TEXTURE2D(_Illum);    SAMPLER(sampler_Illum);
TEXTURE2D(_BumpMap);  SAMPLER(sampler_BumpMap);

CBUFFER_START(UnityPerMaterial)
    float4 _MainTex_ST;
    float4 _Illum_ST;
    float4 _BumpMap_ST;
    half4 _Color;
    half4 _SpecColor;
    half _Shininess;
    half _Emission;
CBUFFER_END

struct UrpifyAttributes
{
    float4 positionOS : POSITION;
    float3 normalOS   : NORMAL;
    float4 tangentOS  : TANGENT;
    float2 uv         : TEXCOORD0;
    float2 lightmapUV : TEXCOORD1;
    UNITY_VERTEX_INPUT_INSTANCE_ID
};

struct UrpifyVaryings
{
    float4 positionCS : SV_POSITION;
    float4 uv         : TEXCOORD0; // xy _MainTex, zw _Illum
    float2 uvBump     : TEXCOORD1;
    float3 positionWS : TEXCOORD2;
    half3  normalWS   : TEXCOORD3;
    half4  tangentWS  : TEXCOORD4; // w: bitangent sign
    half   fogFactor  : TEXCOORD5;
    float2 lightmapUV : TEXCOORD6;
    UNITY_VERTEX_INPUT_INSTANCE_ID
    UNITY_VERTEX_OUTPUT_STEREO
};

UrpifyVaryings UrpifyVert(UrpifyAttributes v)
{
    UrpifyVaryings o = (UrpifyVaryings)0;
    UNITY_SETUP_INSTANCE_ID(v);
    UNITY_TRANSFER_INSTANCE_ID(v, o);
    UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
    VertexPositionInputs pos = GetVertexPositionInputs(v.positionOS.xyz);
    VertexNormalInputs nrm = GetVertexNormalInputs(v.normalOS, v.tangentOS);
    o.positionCS = pos.positionCS;
    o.positionWS = pos.positionWS;
    o.normalWS = nrm.normalWS;
    o.tangentWS = half4(nrm.tangentWS, v.tangentOS.w * GetOddNegativeScale());
    o.uv.xy = TRANSFORM_TEX(v.uv, _MainTex);
    o.uv.zw = TRANSFORM_TEX(v.uv, _Illum);
    o.uvBump = TRANSFORM_TEX(v.uv, _BumpMap);
    o.fogFactor = ComputeFogFactor(pos.positionCS.z);
    o.lightmapUV = v.lightmapUV * unity_LightmapST.xy + unity_LightmapST.zw;
    return o;
}

half3 UrpifyNormal(UrpifyVaryings i)
{
    half3 n = normalize(i.normalWS);
#if defined(URPIFY_BUMP)
    half3 t = SafeNormalize(i.tangentWS.xyz);
    half3 b = cross(n, t) * i.tangentWS.w;
    half3 ts = UnpackNormal(SAMPLE_TEXTURE2D(_BumpMap, sampler_BumpMap, i.uvBump));
    n = normalize(TransformTangentToWorld(ts, half3x3(t, b, n)));
#endif
    return n;
}

// One light as Built-in's legacy Lambert / Blinn-Phong lighting model computes it.
half3 UrpifyLight(Light light, half3 albedo, half gloss, half3 normalWS, half3 viewWS)
{
    half3 color = light.color * (light.distanceAttenuation * light.shadowAttenuation);
    half3 c = albedo * color * max(0, dot(normalWS, light.direction));
#if defined(URPIFY_SPECULAR)
    half3 h = SafeNormalize(light.direction + viewWS);
    float nh = max(0, dot(normalWS, h));
    c += color * _SpecColor.rgb * (pow(nh, _Shininess * 128.0) * gloss);
#endif
    return c;
}

half4 UrpifyFrag(UrpifyVaryings i) : SV_Target
{
    UNITY_SETUP_INSTANCE_ID(i);
    UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(i);
    half4 tex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, i.uv.xy);
    half4 c = tex * _Color;
    half3 albedo = c.rgb;
    half3 emission = c.rgb * SAMPLE_TEXTURE2D(_Illum, sampler_Illum, i.uv.zw).a;
    half gloss = tex.a;
    half3 normalWS = UrpifyNormal(i);
    half3 viewWS = GetWorldSpaceNormalizeViewDir(i.positionWS);

    InputData inputData = (InputData)0;
    inputData.positionWS = i.positionWS;
    inputData.normalWS = normalWS;
    inputData.viewDirectionWS = viewWS;
    inputData.shadowCoord = TransformWorldToShadowCoord(i.positionWS);
    inputData.normalizedScreenSpaceUV = GetNormalizedScreenSpaceUV(i.positionCS);
    inputData.shadowMask = half4(1, 1, 1, 1);
#if defined(LIGHTMAP_ON)
    half3 ambient = SampleLightmap(i.lightmapUV, normalWS);
#else
    half3 ambient = SampleSH(normalWS);
#endif

    half3 color = UrpifyLight(GetMainLight(inputData.shadowCoord), albedo, gloss, normalWS, viewWS);
#if defined(_ADDITIONAL_LIGHTS) || defined(_ADDITIONAL_LIGHTS_VERTEX)
    #if USE_FORWARD_PLUS || USE_CLUSTER_LIGHT_LOOP
    for (uint dirIndex = 0; dirIndex < min(URP_FP_DIRECTIONAL_LIGHTS_COUNT, MAX_VISIBLE_LIGHTS); dirIndex++)
        color += UrpifyLight(GetAdditionalLight(dirIndex, i.positionWS, inputData.shadowMask), albedo, gloss, normalWS, viewWS);
    #endif
    uint pixelLightCount = GetAdditionalLightsCount();
    LIGHT_LOOP_BEGIN(pixelLightCount)
        color += UrpifyLight(GetAdditionalLight(lightIndex, i.positionWS, inputData.shadowMask), albedo, gloss, normalWS, viewWS);
    LIGHT_LOOP_END
#endif
    color += albedo * ambient + emission;
    color = MixFog(color, i.fogFactor);
    return half4(color, 1); // opaque, as Built-in writes alpha 1 for opaque surface shaders
}

// ---------------------------------------------------------------------------------------- shadows and depth

float3 _LightDirection;
float3 _LightPosition;

struct UrpifyDepthVaryings
{
    float4 positionCS : SV_POSITION;
    half3  normalWS   : TEXCOORD0;
    float2 uvBump     : TEXCOORD1;
    half4  tangentWS  : TEXCOORD2;
    UNITY_VERTEX_INPUT_INSTANCE_ID
    UNITY_VERTEX_OUTPUT_STEREO
};

UrpifyDepthVaryings UrpifyDepthVert(UrpifyAttributes v)
{
    UrpifyDepthVaryings o = (UrpifyDepthVaryings)0;
    UNITY_SETUP_INSTANCE_ID(v);
    UNITY_TRANSFER_INSTANCE_ID(v, o);
    UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
    VertexNormalInputs nrm = GetVertexNormalInputs(v.normalOS, v.tangentOS);
    o.positionCS = TransformObjectToHClip(v.positionOS.xyz);
    o.normalWS = nrm.normalWS;
    o.tangentWS = half4(nrm.tangentWS, v.tangentOS.w * GetOddNegativeScale());
    o.uvBump = TRANSFORM_TEX(v.uv, _BumpMap);
    return o;
}

UrpifyDepthVaryings UrpifyShadowVert(UrpifyAttributes v)
{
    UrpifyDepthVaryings o = (UrpifyDepthVaryings)0;
    UNITY_SETUP_INSTANCE_ID(v);
    UNITY_TRANSFER_INSTANCE_ID(v, o);
    float3 positionWS = TransformObjectToWorld(v.positionOS.xyz);
    float3 normalWS = TransformObjectToWorldNormal(v.normalOS);
#if defined(_CASTING_PUNCTUAL_LIGHT_SHADOW)
    float3 lightDirectionWS = normalize(_LightPosition - positionWS);
#else
    float3 lightDirectionWS = _LightDirection;
#endif
    o.positionCS = TransformWorldToHClip(ApplyShadowBias(positionWS, normalWS, lightDirectionWS));
#if UNITY_REVERSED_Z
    o.positionCS.z = min(o.positionCS.z, UNITY_NEAR_CLIP_VALUE);
#else
    o.positionCS.z = max(o.positionCS.z, UNITY_NEAR_CLIP_VALUE);
#endif
    return o;
}

half4 UrpifyShadowFrag(UrpifyDepthVaryings i) : SV_Target { return 0; }

half4 UrpifyDepthFrag(UrpifyDepthVaryings i) : SV_Target { return i.positionCS.z; }

half4 UrpifyDepthNormalsFrag(UrpifyDepthVaryings i) : SV_Target
{
    UNITY_SETUP_INSTANCE_ID(i);
    half3 n = normalize(i.normalWS);
#if defined(URPIFY_BUMP)
    half3 t = SafeNormalize(i.tangentWS.xyz);
    half3 b = cross(n, t) * i.tangentWS.w;
    n = normalize(TransformTangentToWorld(UnpackNormal(SAMPLE_TEXTURE2D(_BumpMap, sampler_BumpMap, i.uvBump)), half3x3(t, b, n)));
#endif
#if defined(_GBUFFER_NORMALS_OCT)
    float2 octNormalWS = PackNormalOctQuadEncode(n);
    return half4(PackFloat2To888(saturate(octNormalWS * 0.5 + 0.5)), 0.0);
#else
    return half4(n, 0.0);
#endif
}
